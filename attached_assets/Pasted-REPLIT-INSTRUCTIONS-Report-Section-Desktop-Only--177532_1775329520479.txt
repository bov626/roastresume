── STEP 3: SCRIPT ───────────────────────────────────────────────────────────
Add this block BEFORE your existing </script> tag (or in a new <script>).
It only runs on large screens so it won't interfere with the mobile scroll.

  // Desktop report scroll (large breakpoint only)
  (function() {
    var section = document.getElementById('report-desktop');
    var track   = document.getElementById('rd-track');
    if (!section || !track) return;
    var NUM = 5;

    function onScroll() {
      if (window.innerWidth < 1024) return;
      var top      = section.getBoundingClientRect().top + window.scrollY;
      var spacer   = section.querySelector('.rd-spacer').offsetHeight;
      var scrollIn = window.scrollY - top;
      if (scrollIn < 0 || scrollIn > spacer) return;
      var p = Math.max(0, Math.min(1, scrollIn / spacer));
      track.style.transform  = 'translateX(-' + (p * (NUM - 1) * 100) + 'vw)';
      track.style.transition = 'none';
    }

    window.addEventListener('scroll', onScroll, { passive: true });
    window.addEventListener('resize', onScroll);
    onScroll();
  })();
